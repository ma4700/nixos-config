\newcommand{\labcompareblock}[7]{%
  \par\smallskip\noindent
  \begin{tabular}{@{}l@{\quad}l@{}}
    \multicolumn{2}{@{}l}{\textbf{Σύγκριση: #1}}\\[0.2em]
    Πειραματική τιμή: & \labvalue{#3}{#2}\\
    Αναμενόμενη τιμή: & \labvalue{#4}{#2}\\
    Σχετικό σφάλμα:   & $\varepsilon = \labpercent{#5}$
                        \quad (προβλεπόμενο: $\le \SI{#6}{\percent}$)\\
    Αξιολόγηση:       & \ifnum#7=1 \textbf{εκτός ορίου}\else εντός ορίου\fi
  \end{tabular}\par\smallskip}

\newcommand{\labsummaryrow}[8]{%
  #1 & #2 & \labvalue{#4}{#3} & \labvalue{#5}{#3} & \labpercent{#6}
  & \SI{#7}{\percent} & \ifnum#8=1 \textbf{εκτός}\else εντός\fi \\}

      \midrule
      \labsummaryrows
      \bottomrule
    \end{tabular}
  \end{table}}

\ExplSyntaxOn
\fp_new:N   \l__lab_err_fp
\bool_new:N \l__lab_out_bool
\bool_new:N \g__lab_step_out_bool
\tl_new:N   \g__lab_rows_tl

\NewDocumentCommand \labstep { m }
  {
    \bool_gset_false:N \g__lab_step_out_bool
    \refstepcounter { labstep }
    \labstepheading {#1}
  }

\NewDocumentCommand \labcompare { O{} m m m m }
  {
    \fp_set:Nn \l__lab_err_fp { abs( (#3) - (#4) ) / abs(#4) * 100 }
    \fp_compare:nNnTF \l__lab_err_fp > {#5}
      {
        \bool_set_true:N  \l__lab_out_bool
        \bool_gset_true:N \g__lab_step_out_bool
      }
      { \bool_set_false:N \l__lab_out_bool }
    \use:x
      {
        \exp_not:N \labcompareblock
          { \exp_not:n {#2} } { \exp_not:n {#1} }
          { \exp_not:n {#3} } { \exp_not:n {#4} }
          { \fp_use:N \l__lab_err_fp } { \exp_not:n {#5} }
          { \bool_if:NTF \l__lab_out_bool { 1 } { 0 } }
      }
    \tl_gput_right:Nx \g__lab_rows_tl
      {
        \exp_not:N \labsummaryrow
          { \int_use:c { c@labstep } }
          { \exp_not:n {#2} } { \exp_not:n {#1} }
          { \exp_not:n {#3} } { \exp_not:n {#4} }
          { \fp_use:N \l__lab_err_fp } { \exp_not:n {#5} }
          { \bool_if:NTF \l__lab_out_bool { 1 } { 0 } }
      }
  }

\NewExpandableDocumentCommand \labifout { +m +m }

\NewExpandableDocumentCommand \labsummaryrows { }

\NewDocumentCommand \labsummary { }
